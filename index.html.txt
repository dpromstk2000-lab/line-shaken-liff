    // 回答データを格納するオブジェクト
    let answers = {
        name: '',
        shaken: '',
        purpose: '',
        carType: '',
        timing: '',
        requests: []
    };

    // 初期化：LIFFの起動とユーザー情報の取得
    document.addEventListener("DOMContentLoaded", function() {
        liff.init({ liffId: "ユーザー毎のLIFF_IDをここに設定" })
            .then(() => {
                console.log("LIFF 初期化成功");
                // LINEにログインしていればプロフィール（名前）を取得して自動入力
                if (liff.isLoggedIn()) {
                    liff.getProfile().then(profile => {
                        document.getElementById('userName').value = profile.displayName;
                    }).catch(err => console.error("プロフィール取得失敗", err));
                }
            })
            .catch((err) => {
                console.error("LIFF 初期化失敗", err);
            });
    });

    // ステップ1（名前・車検年月）の保存と次へ
    function saveStep1() {
        const nameVal = document.getElementById('userName').value.trim();
        const shakenVal = document.getElementById('shakenDate').value;

        if(!nameVal) {
            alert('お名前を入力してください。');
            return;
        }
        
        answers.name = nameVal;
        
        if(shakenVal) {
            // 「2026-05」を「2026年05月」の形式に変換
            const parts = shakenVal.split('-');
            answers.shaken = `${parts[0]}年${parts[1]}月`;
        } else {
            answers.shaken = '不明・未入力';
        }

        nextStep(2);
    }

    // 単一選択ボタンの処理
    function selectSingle(currentStep, value) {
        if(currentStep === 2) answers.purpose = value;
        if(currentStep === 3) answers.carType = value;
        if(currentStep === 4) answers.timing = value;

        const stepDiv = document.getElementById(`step${currentStep}`);
        const buttons = stepDiv.querySelectorAll('.option-btn');
        buttons.forEach(btn => btn.classList.remove('selected'));
        event.currentTarget.classList.add('selected');

        setTimeout(() => {
            nextStep(currentStep + 1);
        }, 200);
    }

    // 複数選択ボタンの処理
    function toggleMulti(element, value) {
        element.classList.toggle('selected');
        const index = answers.requests.indexOf(value);
        if (index > -1) {
            answers.requests.splice(index, 1);
        } else {
            answers.requests.push(value);
        }
    }

    // 画面遷移コントロール
    function nextStep(stepNum) {
        document.querySelectorAll('.step').forEach(step => step.classList.remove('active'));
        document.getElementById(`step${stepNum}`).classList.add('active');
        
        const percent = (stepNum / 6) * 100;
        document.getElementById('progressBar').style.width = `${percent}%`;
    }

    function prevStep(stepNum) {
        document.querySelectorAll('.step').forEach(step => step.classList.remove('active'));
        document.getElementById(`step${stepNum}`).classList.add('active');
        
        const percent = (stepNum / 6) * 100;
        document.getElementById('progressBar').style.width = `${percent}%`;
    }

    // LINEへデータを送信して画面を閉じる
    function submitData() {
        const reqText = answers.requests.length > 0 ? answers.requests.join(', ') : '特になし';
        
        // LINEのトーク画面に自動投稿するテキストを作成
        const messageText = `【整備・見積もり診断リクエスト】\n` +
                            `■お名前: ${answers.name} 様\n` +
                            `■次回の車検: ${answers.shaken}\n` +
                            `■ご相談内容: ${answers.purpose}\n` +
                            `■お車のタイプ: ${answers.carType}\n` +
                            `■ご希望の時期: ${answers.timing}\n` +
                            `■ご要望・気になる点: ${reqText}\n` +
                            `予約の空き状況や、概算の見積もりを教えてください！`;

        // LIFF経由でメッセージを送信
        if (liff.isInClient()) {
            liff.sendMessages([{
                type: 'text',
                text: messageText
            }]).then(() => {
                liff.closeWindow();
            }).catch((err) => {
                alert('メッセージの送信に失敗しました: ' + err);
            });
        } else {
            alert("LINEアプリ以外で開いているため、本来はここでチャットに以下の内容が送信されて画面が閉じます：\n\n" + messageText);
        }
    }</script></body></html>
